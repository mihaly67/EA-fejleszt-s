void CleanChartCompletely(long chart_id) {
    // 1. Töröl minden objektumot (vonalak, nyilak, szövegek, panelek)
    ObjectsDeleteAll(chart_id, -1, -1);
    ObjectsDeleteAll(chart_id, 0, -1); // Főablak

    // 2. Eltávolít minden indikátort (kivéve magát az EA-t, ha ez az EA)
    int windows = (int)ChartGetInteger(chart_id, CHART_WINDOWS_TOTAL);
    for(int w=0; w<windows; w++) {
        int ind_total = ChartIndicatorsTotal(chart_id, w);
        // Hátrafelé iterálunk, hogy a törlés ne keverje össze az indexeket
        for(int i=ind_total-1; i>=0; i--) {
            string ind_name = ChartIndicatorName(chart_id, w, i);
            ChartIndicatorDelete(chart_id, w, ind_name);
        }
    }

    // 3. Trade history kikapcsolása a charton (ne mutassa a nyilakat a múltból)
    ChartSetInteger(chart_id, CHART_SHOW_TRADE_HISTORY, false);

    // Frissítés
    ChartRedraw(chart_id);
}
